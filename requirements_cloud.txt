# Cloud-instance bootstrap dependencies (Lambda Cloud + others).
# Use with --system-site-packages venv so we DON'T override the image's
# pre-installed torch (Lambda's image ships torch 2.7.0 + CUDA 12.8; pip
# would otherwise pull a newer torch that requires a newer CUDA driver).
#
# Pinning intent:
#   * numpy/scipy: project uses recent versions, but compatible with
#     system torch's numpy ABI. Floors not ceilings.
#   * faiss-cpu: substrate testbed work; not strictly required by every
#     experiment but cheap to install.
#   * fastapi/pydantic/httpx: hdlab_service Pattern B server + tests
#   * cryptography: Ed25519 deletion certificates
#   * pytest: test suites
#
# torch INTENTIONALLY OMITTED -- use system torch via --system-site-packages.
# Add torch back here only if you've also pinned it to the exact version
# matching the image's CUDA driver.

# numpy: pin to <2 because Lambda's image-pre-installed torch 2.7.0 was
# compiled against numpy 1.x ABI. Installing numpy 2.x breaks
# torch.from_numpy / torch <-> numpy conversions at runtime even though
# `import torch` still succeeds (warning: 'Failed to initialize NumPy:
# _ARRAY_API not found'). Substrate experiments hit this constantly.
numpy>=1.24,<2.0
scipy>=1.10,<1.14
faiss-cpu>=1.7.4
pydantic>=2.0
fastapi>=0.100
httpx>=0.24
cryptography>=41.0
